option(SKIFREE_BUILD_UI "Build SDL window/input and raw GLES2 renderer" ON)
option(SKIFREE_WEBOS "Use a verified webOS ARM cross-toolchain" OFF)
option(SKIFREE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(SKIFREE_STRICT "Treat compiler warnings as errors" ON)

# Adapted from Spice Home C++: retain its real cross-toolchain/header gates.
if(SKIFREE_WEBOS)
  if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^[aA][rR][mM]")
    message(FATAL_ERROR "SKIFREE_WEBOS requires a real ARM cross-toolchain, not a host executable")
  endif()
endif()
function(skifree_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    if(SKIFREE_STRICT)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
    if(SKIFREE_STRICT)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(SKIFREE_SANITIZE)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endfunction()

add_library(skifree-game src/classic/core.cpp src/recreation/game.cpp src/platform/core.cpp)
target_include_directories(skifree-game PUBLIC include)
# The reconstructed core retains old C idioms; compile its signed 32-bit
# arithmetic as the original x86 code intended, without aliasing assumptions.
if(NOT MSVC)
  set_source_files_properties(src/classic/core.cpp PROPERTIES COMPILE_OPTIONS "-fwrapv;-fno-strict-aliasing")
endif()
if(SKIFREE_SANITIZE AND NOT MSVC)
  target_compile_options(skifree-game PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(skifree-game PUBLIC -fsanitize=address,undefined)
endif()

if(SKIFREE_BUILD_UI)
  find_package(PNG REQUIRED)
  find_package(Threads REQUIRED)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
  add_executable(skifree-native src/main.cpp src/recreation/render.cpp src/platform/runtime.cpp src/platform/render.cpp)
  target_include_directories(skifree-native PRIVATE include compat ${SDL2_INCLUDE_DIR})
  target_link_libraries(skifree-native PRIVATE skifree-game PNG::PNG Threads::Threads ${CMAKE_DL_LIBS})
  if(SKIFREE_WEBOS)
    target_compile_definitions(skifree-native PRIVATE SKIFREE_WEBOS=1)
  endif()
  set_target_properties(skifree-native PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
  skifree_warnings(skifree-native)
  install(TARGETS skifree-native RUNTIME DESTINATION .)
  # Ship only the sprite aliases recorded in the audited source manifest.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/manifest.json" asset_manifest)
  string(JSON asset_count LENGTH "${asset_manifest}" files)
  math(EXPR asset_last "${asset_count} - 1")
  foreach(asset_index RANGE 0 ${asset_last})
    string(JSON asset_name GET "${asset_manifest}" files ${asset_index} name)
    if(NOT asset_name MATCHES "^[a-z0-9_]+[.]png$")
      message(FATAL_ERROR "Invalid bundled asset filename")
    endif()
    install(FILES "assets/${asset_name}" DESTINATION assets)
  endforeach()
  install(FILES assets/README.md assets/manifest.json DESTINATION assets)
  install(FILES appinfo.json icon.png DESTINATION .)
  install(FILES LICENSE NOTICE.md DESTINATION .)
  install(FILES state/README.md DESTINATION state)
  install(FILES src/classic/PROVENANCE.md DESTINATION . RENAME CLASSIC-PROVENANCE.md)
endif()
